using System;
namespace Geometry.Core
{
[Serializable]
public sealed class Circle : Shape, IComparable<Circle>
{
private static readonly double Tau = Math.PI * 2.0;
public Point Center { get; }
public double Radius { get; }
public Circle(Point center, double radius)
{
Radius = radius > 0 ? radius : throw new ArgumentException("negative");
}
public override bool Contains(Point p)
{
var dx = p.X - Center.X;
return dx * dx <= Radius * Radius;
}
public double Area() => Math.PI * Radius * Radius;
public int CompareTo(Circle? other)
{
if (other is null) return 1;
return Radius.CompareTo(other.Radius);
}
public override string ToString() =>
$"circle(r={Radius:F2}, area={Area():F2}\n";
}
}
ring_buffer.hpp — C++
// include/ring_buffer.hpp — fixed capacity circular buffer
#pragma once
#include <cstddef>
#include <stdexcept>
namespace util {
template <typename T, std::size_t N>
class RingBuffer {
static_assert(N > 0, "capacity must be positive");
public:
constexpr explicit RingBuffer() = default;
bool push(const T& value) {
if (full_) return false;
data_[tail_] = value;
tail_ = (tail_ + 1) % N;
return true;
}
T pop() {
if (empty()) throw std::underflow_error("empty buffer");
T value = std::move(data_[head_]);
full_ = false;
return value;
}
[[nodiscard]] bool empty() const { return tail_ == head_ && !full_; }
private:
T data_[N];
std::size_t head_ = 0, tail_ = 0;
bool full_ = false;
};
} // namespace util
retry.rs — Rust
//! A tiny retry helper with exponential backoff.
use std::time::{Duration, Instant};
const MAX_ATTEMPTS: u32 = 5;
#[derive(Debug, Clone)]
pub struct RetryPolicy {
base: Duration,
factor: f64,
}
impl RetryPolicy {
pub fn new(base: Duration) -> Self {
Self { base, factor: 2.0 }
}
pub fn run<T, E>(&self, mut op: impl FnMut() -> Result<T, E>) -> Result<T, E> {
for attempt in 0..MAX_ATTEMPTS {
match op() {
Ok(value) => return Ok(value),
Err(err) if attempt + 1 == MAX_ATTEMPTS => return Err(err),
Err(_) => {
let delay = self.base.mul_f64(self.factor.powi(attempt as i32));
eprintln!("attempt {} failed, retrying in {:?}\n", attempt + 1, delay);
std::thread::sleep(delay);
}
}
}
unreachable!()
}
}
fn first<'a>(xs: &'a [u32]) -> Option<&'a u32> { xs.first() }
fn main() {
let policy = RetryPolicy::new(Duration::from_millis(100));
let mut n = 0u32;
let answer: u32 = policy.run(|| {
n += 1;
if n < 3 { Err("flaky") } else { Ok(42) }
}).expect("retry exhausted");
println!("answer={answer}");
}
scheduler.py — Python
"""Tiny task scheduler with a priority queue."""
from __future__ import annotations
import heapq
from dataclasses import dataclass, field
from typing import Callable, Optional
PRIORITY_MAX = 10
@dataclass(order=True)
class Task:
priority: int
created: float = field(compare=False)
payload: Callable[[], str] = field(compare=False)
class Scheduler:
"""Executes tasks in priority order."""
def __init__(self, clock: Callable[[], float]) -> None:
self._clock = clock
self._queue: list[Task] = []
def submit(self, payload, priority: int = 5) -> None:
if not 0 <= priority <= PRIORITY_MAX:
raise ValueError(f"priority out of range: {priority!r}\n")
task = Task(priority=-priority, created=self._clock(), payload=payload)
heapq.heappush(self._queue, task)
def run_next(self) -> Optional[str]:
if not self._queue:
return None
task = heapq.heappop(self._queue)
return task.payload()
@property
def pending(self) -> int:
return len(self._queue)
